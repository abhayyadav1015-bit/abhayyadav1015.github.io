        async function initializeFirebase() {
            if (!firebaseConfig.apiKey) {
                console.error("Firebase config is missing. Running in mock mode.");
                isAuthReady = true;
                userId = "mock-user-123";
                currentPage = 'Home';
                updateContent();
                return;
            }

            try {
                firebase.setLogLevel('Debug');
                const app = firebase.initializeApp(firebaseConfig);
                auth = firebase.getAuth(app);
                db = firebase.getFirestore(app);

                if (initialAuthToken) {
                    await firebase.signInWithCustomToken(auth, initialAuthToken);
                } else {
                    await firebase.signInAnonymously(auth);
                }
                
                firebase.onAuthStateChanged(auth, async (user) => {
                    if (user) {
                        userId = user.uid;
                        profileData.email = user.email || 'N/A';
                        await loadProfileData(userId);
                        if (currentPage === 'Auth' || !isAuthReady) {
                             currentPage = 'Home';
                        }
                    } else {
                        userId = null;
                        currentPage = 'Auth';
                    }
                    isAuthReady = true;
                    updateContent();
                });

            } catch (error) {
                console.error("Firebase Initialization Error:", error);
                isAuthReady = true;
                currentPage = 'Auth';
                updateContent();
            }
        }

        function getProfileDocRef(uid) {
            // MODIFIED: This now points to a single, shared public document
            // The 'uid' parameter is ignored to ensure everyone edits the same document.
            return firebase.doc(db, 'artifacts', appId, 'public', 'data', 'sharedProfile', 'profile');
        }

        async function loadProfileData(uid) {
            try {
                const profileRef = getProfileDocRef(uid);
                const docSnap = await firebase.getDoc(profileRef);

                if (docSnap.exists()) {
                    const loadedData = docSnap.data();
                    profileData = { ...profileData, ...loadedData };
                } else {
                    // Initialize document if it doesn't exist
                    profileData.email = auth.currentUser?.email || 'N/A'; // Ensure email is set on creation
                    await saveProfileData(uid);
                }
            } catch (e) {
                console.error("Error loading profile data: ", e);
            }
        }

        async function saveProfileData(uid) {
            try {
                const profileRef = getProfileDocRef(uid);
                await firebase.setDoc(profileRef, {
                    name: profileData.name,
                    email: profileData.email,
                    phone: profileData.phone,
                    gender: profileData.gender, 
                    address: profileData.address,
                }, { merge: true });
            } catch (e) {
                console.error("Error saving profile data: ", e);
            }
        }

        async function generateOutfitImage(prompt) { // Renamed from generateTrendImage
            const payload = {
                instances: [{ prompt: prompt }],
                parameters: { "sampleCount": 1 }
            };
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/imagen-3.0-generate-002:predict?key=${apiKey}`;
            let retryCount = 0;
            const maxRetries = 3;
            let response;
            while (retryCount < maxRetries) {
                try {
                    response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    if (response.ok) {
                        break;
                    } else if (response.status === 429) {
                        const delay = Math.pow(2, retryCount) * 1000 + Math.random() * 1000;
                        await new Promise(resolve => setTimeout(resolve, delay));
                        retryCount++;
                    } else {
                        throw new Error(`API returned status ${response.status}`);
                    }
                } catch (error) {
                    console.error("Fetch error:", error);
                    return "https://placehold.co/400x500/FCA5A5/fff?text=Generation+Failed";
                }
            }
            if (!response || !response.ok) {
                return "https://placehold.co/400x500/FCA5A5/fff?text=Generation+Failed";
            }
            const result = await response.json();
            const base64Data = result?.predictions?.[0]?.bytesBase64Encoded;
            if (base64Data) {
                return `data:image/png;base64,${base64Data}`;
            } else {
                return "https://placehold.co/400x500/FCA5A5/fff?text=No+Image+Data";
            }
        }

        async function fetchLatestTrends() {
            if (isTrendsLoading) return;
            isTrendsLoading = true;
            latestTrends = []; // Clear previous trends
            currentTrendIndex = 0; // Reset index
            updateContent();

            const userGender = profileData.gender || 'All';
            const genderQuery = userGender === 'All' ? "unisex fashion" : `${userGender.toLowerCase()}'s fashion`;

            const systemPrompt = "You are a world-class fashion expert and trend-spotter, like a top editor for Vogue or a major fashion influencer. Your goal is to find the absolute latest, most popular fashion trends emerging right now on social media (like TikTok, Instagram, pinterest.com) and fashion blogs.";
            
            const userQuery = `Identify 3 distinct, cutting-edge ${genderQuery} trends for this season. For each trend, provide: \n1. A catchy **Trend Name** (e.g., 'Cyber-Punk Utility', 'Retro Academia').\n2. A concise 1-2 sentence **Description** of the style.\n3. A list of 3-4 **Key Items** that define the look.\n4. A **Visual Prompt** for an AI image generator to create a sample outfit (e.g., 'A person wearing an oversized distressed leather jacket, cargo pants with many pockets, and chunky platform boots, cyberpunk style').`;

            const schema = {
              "type": "OBJECT",
              "properties": {
                "trends": {
                  "type": "ARRAY",
                  "items": {
                    "type": "OBJECT",
                    "properties": {
                      "trendName": { "type": "STRING" },
                      "description": { "type": "STRING" },
                      "keyItems": { "type": "ARRAY", "items": { "type": "STRING" } },
                      "visualPrompt": { "type": "STRING" }
                    },
                    "required": ["trendName", "description", "keyItems", "visualPrompt"]
                  }
                }
              },
              "required": ["trends"]
            };

            const payload = {
                contents: [{ parts: [{ text: userQuery }] }],
                tools: [{ "google_search": {} }],
                systemInstruction: {
                    parts: [{ text: systemPrompt }]
                },
                generationConfig: {
                  responseMimeType: "application/json",
                  responseSchema: schema
                }
            };
            
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    throw new Error(`API returned status ${response.status}`);
                }

                const result = await response.json();
                const candidate = result.candidates?.[0];
                
                if (candidate && candidate.content?.parts?.[0]?.text) {
                    const jsonText = candidate.content.parts[0].text;
                    const parsedJson = JSON.parse(jsonText);

                    if (parsedJson.trends && parsedJson.trends.length > 0) {
                        // We have trends, now generate images for them
                        let generatedTrends = [];
                        for (const trend of parsedJson.trends) {
                            const imageData = await generateOutfitImage(trend.visualPrompt); // Use renamed function
                            generatedTrends.push({ ...trend, imageData });
                        }
                        latestTrends = generatedTrends;
                    } else {
                        throw new Error("No trends found in API response.");
                    }
                } else {
                    throw new Error("Invalid API response structure.");
                }

            } catch (error) {
                console.error("Error fetching latest trends:", error);
                showModal("Error", "Could not fetch latest trends. Please try again.");
                latestTrends = []; // Ensure it's empty on error
            }

            isTrendsLoading = false;
            updateContent();
        }

        function updateSelectorsFromCart() {
             const topSelectCart = document.getElementById('top-select-cart');
             const bottomSelectCart = document.getElementById('bottom-select-cart');
             const topSelect = document.getElementById('top-select');
             const bottomSelect = document.getElementById('bottom-select');

             // Set main selectors to cart selections if they exist
             if (topSelectCart && topSelectCart.value) {
                const normalizedValue = topSelectCart.value.replace(' (In Cart)', '').trim();
                const matchingOption = [...topSelect.options].find(opt => opt.value.startsWith(normalizedValue));
                if(matchingOption) topSelect.value = matchingOption.value;
             }
             
             if (bottomSelectCart && bottomSelectCart.value) {
                 const normalizedValue = bottomSelectCart.value.replace(' (In Cart)', '').trim();
                 const matchingOption = [...bottomSelect.options].find(opt => opt.value.startsWith(normalizedValue));
                 if(matchingOption) bottomSelect.value = matchingOption.value;
             }

             handleVisualization('select');

         function updateContent() {
            const contentArea = document.getElementById('content-area');
            const mainHeader = document.getElementById('main-header');
            const bottomNav = document.getElementById('bottom-nav');
            const navElements = document.querySelectorAll('.nav-btn');
            
            // Clean up 3D scene if navigating away (REMOVED)
            /*
            if (currentPage !== 'Visualization') {
                ... (REMOVED) ...
            }
            */
 
            if (!isAuthReady) {
                if (contentArea) contentArea.innerHTML = '<div class="text-center py-20"><svg class="animate-spin h-10 w-10 text-violet-600 mx-auto" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24"><circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle><path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path></svg><p class="mt-3 text-lg text-slate-600">Loading Trendy Nest...</p></div>';
                if (mainHeader) mainHeader.classList.add('hidden');
                if (bottomNav) bottomNav.classList.add('hidden');
                return;
            }
 
            if (!userId) {
                currentPage = (currentPage === 'Register' ? 'Register' : 'Login');
                if (contentArea) contentArea.innerHTML = renderAuthPage();
                if (mainHeader) mainHeader.classList.add('hidden');
                if (bottomNav) bottomNav.classList.add('hidden');
                return;
            }
 
            if (mainHeader) mainHeader.classList.remove('hidden');
            if (bottomNav) bottomNav.classList.remove('hidden');
            let contentHTML = '';
 
            // Render the page content based on the current state
            switch (currentPage) {
                case 'Search': contentHTML = renderSearchPage(); break;
                case 'Visualization': contentHTML = renderVisualizationPage(); break;
                case 'Buy': contentHTML = renderBuyPage(); break;
                case 'Profile': contentHTML = renderProfilePage(); break;
                case 'Home': default: contentHTML = renderHomePage(); break;
            }
            if (contentArea) contentArea.innerHTML = contentHTML;
            
            // AFTER content is in DOM, initialize 3D scene if on that page (REMOVED)
            /*
            if (currentPage === 'Visualization') {
                initThreeJSScene();
            }
            */

            // Update navigation button styles
            navElements.forEach(btn => {
                btn.classList.remove('nav-active');
                if (btn.dataset.page === currentPage) {
                    btn.classList.add('nav-active');
                }
            });
            
            // Set up form submission listeners after content is loaded
            if (currentPage === 'Login' || currentPage === 'Register') {
                document.getElementById('auth-form').onsubmit = (e) => {
                    e.preventDefault();
                    const email = document.getElementById('auth-email').value;
                    const password = document.getElementById('auth-password').value;
                    if (currentPage === 'Login') {
                        handleLogin(email, password);
                    } else {
                        handleRegister(email, password);
                    }
                };
            } else if (currentPage === 'Profile') {
                 document.getElementById('personal-edit-form').onsubmit = async (e) => {
                    e.preventDefault();
                    profileData.name = document.getElementById('edit-name').value;
                    profileData.phone = document.getElementById('edit-phone').value;
                    profileData.gender = document.getElementById('edit-gender').value; 
                    await saveProfileData(userId);
                    toggleEdit('personal');
                    updateContent(); // Re-render to show updated details
                };
            }
        }

                // Check if already in cart

                     showModal("Already in Cart", `<span class="font-bold text-violet-700">${product.name}</span> is already in your shopping cart.`);

                }
            } else {
                showModal("Error", "Could not find the selected product.");
            }
        };
